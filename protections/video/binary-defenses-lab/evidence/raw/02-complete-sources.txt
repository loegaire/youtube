===== src/canary_probe.c =====
     1	#include <inttypes.h>
     2	#include <stdint.h>
     3	#include <stdio.h>
     4	
     5	__attribute__((noinline))
     6	static void probe(void) {
     7	    char buffer[16] = {0};
     8	    uintptr_t tls_guard = 0;
     9	    uintptr_t frame_copy = 0;
    10	
    11	#if defined(__x86_64__)
    12	    __asm__ volatile("mov %%fs:0x28, %0" : "=r"(tls_guard));
    13	    __asm__ volatile("mov -8(%%rbp), %0" : "=r"(frame_copy));
    14	#endif
    15	
    16	    printf("buffer     %p\n", (void *)buffer);
    17	    printf("tls-guard  0x%016" PRIxPTR "\n", tls_guard);
    18	    printf("frame-copy 0x%016" PRIxPTR "\n", frame_copy);
    19	}
    20	
    21	int main(void) {
    22	    probe();
    23	    return 0;
    24	}
